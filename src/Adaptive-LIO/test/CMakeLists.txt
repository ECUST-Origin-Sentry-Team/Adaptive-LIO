cmake_minimum_required(VERSION 3.16)
project(adaptive_lio_registration_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Ceres REQUIRED)
find_package(Eigen3 REQUIRED)

target_include_directories(test_registration PRIVATE ../src ../thirdparty)
target_link_libraries(test_registration PRIVATE Ceres::ceres Eigen3::Eigen)
add_test(NAME registration COMMAND test_registration)
